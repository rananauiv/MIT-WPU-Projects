import re
import getpass


COMMON_PASSWORDS = {
    "password", "123456", "12345678", "123456789",
    "qwerty", "abc123", "password123", "admin",
    "letmein", "welcome", "iloveyou"
}


def check_password(password):
    score = 0
    feedback = []

    if password.lower() in COMMON_PASSWORDS:
        return "Very Weak", 0, ["Commonly used password."]

    if len(password) >= 8:
        score += 1
    else:
        feedback.append("Use at least 8 characters.")

    if len(password) >= 12:
        score += 1

    if len(password) >= 16:
        score += 1

    if re.search(r"[a-z]", password):
        score += 1
    else:
        feedback.append("Add lowercase letters.")

    if re.search(r"[A-Z]", password):
        score += 1
    else:
        feedback.append("Add uppercase letters.")

    if re.search(r"\d", password):
        score += 1
    else:
        feedback.append("Add numbers.")

    if re.search(r"[^A-Za-z0-9]", password):
        score += 1
    else:
        feedback.append("Add symbols.")

    if re.search(r"(.)\1\1", password):
        score -= 1
        feedback.append("Avoid repeated characters.")

    score = max(0, min(score, 7))

    if score <= 2:
        strength = "Very Weak"
    elif score <= 4:
        strength = "Weak"
    elif score == 5:
        strength = "Moderate"
    elif score == 6:
        strength = "Strong"
    else:
        strength = "Very Strong"

    return strength, score, feedback


try:
    password = getpass.getpass("Password: ")

    strength, score, feedback = check_password(password)

    print("\n" + "=" * 40)
    print("PASSWORD STRENGTH")
    print("=" * 40)

    print(f"Strength : {strength}")
    print(f"Score    : {score}/7")

    if feedback:
        print("\nSuggestions:")
        for item in feedback:
            print(f"- {item}")
    else:
        print("\nNo obvious weaknesses detected.")

    print("=" * 40)

except KeyboardInterrupt:
    print("\n\nCancelled.")
