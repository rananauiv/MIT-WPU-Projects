# 1. Largest and smallest element
numbers = (12, 45, 7, 89, 23, 56, 3, 91, 34, 18)

largest = numbers[0]
smallest = numbers[0]

for num in numbers:
    if num > largest:
        largest = num
    if num < smallest:
        smallest = num

print("\n1. Largest:", largest)
print("Smallest:", smallest)

# 39. Find all pairs whose sum equals a given number

# 40. Sum of each row and each column
matrix = (
    (10, 20, 30),
    (40, 50, 60),
    (70, 80, 90)
)

print("\n40. Row sums:")

for i in range(len(matrix)):
    row_sum = sum(matrix[i])
    print("Row", i + 1, "sum:", row_sum)

print("Column sums:")

for j in range(len(matrix[0])):
    column_sum = 0

    for i in range(len(matrix)):
        column_sum += matrix[i][j]

    print("Column", j + 1, "sum:", column_sum)
