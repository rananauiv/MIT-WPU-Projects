students = []

# Subject name : maximum attendance
subjects = {
    "Mathematics": 100,
    "Physics": 70,
    "Chemistry": 80,
    "biology": 50
}

num_students = 3

for student_number in range(1, num_students + 1):

    scores = []

    print(f"\nStudent {student_number}")

    for subject, maximum in subjects.items():

        while True:
            try:
                score = int(
                    input(f"Enter attendance for {subject} (0-{maximum}): ")
                )

                if 0 <= score <= maximum:
                    scores.append(score)
                    break

                print(f"Invalid attendance. Enter a value between 0 and {maximum}.")

            except ValueError:
                print("Invalid input. Enter a whole number.")




print("\nStudent Data:")
print(students)
